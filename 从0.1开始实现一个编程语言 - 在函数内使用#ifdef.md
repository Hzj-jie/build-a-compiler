# 从0.1开始实现一个编程语言 - 在函数内使用#ifdef

严格来说这个标题是不准确的，所需要的改动是让#ifdef或者类似的类宏语句可以包裹一个语句而不只是类、函数等。之所以称其为类宏语句，是因为#ifdef本质是语句，bstyle/b2style/b3style只需要一次编译即可处理这些语句，而没有c/c++的预处理过程。

现在对#ifdef或类似的语句的使用类似于下面的例子，

```cpp
#ifdef B3STYLE
string _str_mid(string s, int i, int l) {
  string r;
  logic "cut_len r s i l";
  return r;
}
#else
string _str_mid(string s, int i, int l) {
  string r;
  logic "cut_len b2style__r b2style__s b2style__i b2style__l";
  return r;
}
#endif

string str_mid(string s, int i, int l) {
  ::assert(__STATEMENT__ + "@" + __FILE__, (i >= 0) && (l >= 0));
  return _str_mid(s, i, l);
}

```

虽然不能说太糟糕，但是重复诸如函数定义等是没有什么意义的。理想的情况应该是如下，

```cpp
string str_mid(string s, int i, int l) {
  ::assert(__STATEMENT__ + "@" + __FILE__, (i >= 0) && (l >= 0));
  string r;
#ifdef B3STYLE
  logic "cut_len r s i l";
#else #ifdef B2STYLE
  logic "cut_len b2style__r b2style__s b2style__i b2style__l";
#else
  // TODO: Trigger an #error.
#endif
#endif
  return r;
}

```

很显然可以省不少代码。

#ifdef等语句的处理本身并不关心其中包裹的是什么，只是简单的传递给其它的语句处理器，[osi.net/service/compiler/scope/scope.define.vb#L109](https://github.com/Hzj-jie/osi.net/blob/8b75a8979d2d9ff3f7934da3184c02e46bfb20ca/service/compiler/scope/scope.define.vb#L109)

```vb.net
Private Function build(ByVal n As typed_node,
                       ByVal o As WRITER) As Boolean Implements code_gen(Of WRITER).build
    assert(Not n Is Nothing)
    assert(n.child_count() >= 3)
    For i As UInt32 = 0 To n.child(1).child_count() - uint32_1
        Dim s As String
        If i = n.child(1).child_count() - uint32_1 Then
            s = n.child(1).last_child().word().str()
        Else
            s = n.child(1).child(i).child(0).word().str()
        End If
        If Not bypass(s) Then
            For j As UInt32 = 2 To n.child_count() - uint32_3
                If Not code_gen_of(n.child(j)).build(o) Then
                    Return False
                End If
            Next
            Return n.child(n.child_count() - uint32_2).type_name.Equals("delse-wrapped") OrElse
                   code_gen_of(n.child(n.child_count() - uint32_2)).build(o)
        End If
    Next
    If n.child(n.child_count() - uint32_2).type_name.Equals("delse-wrapped") Then
        For j As UInt32 = uint32_1 To n.child(n.child_count() - uint32_2).child_count() - uint32_1
            If Not code_gen_of(n.child(n.child_count() - uint32_2).child(j)).build(o) Then
                Return False
            End If
        Next
        Return True
    End If
    Return True
End Function

```

简单来说就是让#ifdef等可以包含段落，同时段落也可以包含#ifdef等语句。

测试一下，[service/tests/compiler/b2style/self-test/ifdef-sentence.txt](https://github.com/Hzj-jie/osi.net/blob/master/service/tests/compiler/b2style/self-test/ifdef-sentence.txt)

```cpp
#include <b2style.h>
#include <testing.h>

void main() {
#ifdef NO
  b2style::testing::assert_true(false);
#else
  b2style::testing::assert_true(true);
#endif

#ifndef NO
  b2style::testing::assert_true(true);
#else
  b2style::testing::assert_true(false);
#endif

#ifdef NO {
  b2style::testing::assert_true(false);
} #else {
  b2style::testing::assert_true(true);
} #endif

#ifndef NO {
  b2style::testing::assert_true(true);
} #else {
  b2style::testing::assert_true(false);
} #endif

#ifdef B2STYLE || B3STYLE
  b2style::testing::finished();
#endif
}

```
